# ========================================
# TERMINAL COMMANDS FOR INFRASTRUCTURE
# ========================================

This file contains all the terminal commands you need to interact with
the infrastructure services running in Docker.

## ========================================
## CONTAINER MANAGEMENT
## ========================================

# Start all infrastructure
docker-compose -f docker-compose-infrastructure.yml up -d

# Stop all infrastructure
docker-compose -f docker-compose-infrastructure.yml down

# Stop and remove volumes (DELETES ALL DATA!)
docker-compose -f docker-compose-infrastructure.yml down -v

# View running containers
docker-compose -f docker-compose-infrastructure.yml ps

# View logs for all services
docker-compose -f docker-compose-infrastructure.yml logs -f

# View logs for specific service
docker logs -f hotel-postgres
docker logs -f hotel-mongo
docker logs -f hotel-redis
docker logs -f hotel-kafka
docker logs -f hotel-zookeeper
docker logs -f hotel-rabbitmq

# Restart a specific service
docker restart hotel-postgres
docker restart hotel-mongo
docker restart hotel-redis
docker restart hotel-kafka
docker restart hotel-zookeeper
docker restart hotel-rabbitmq


## ========================================
## POSTGRESQL COMMANDS
## ========================================

# Access PostgreSQL CLI
docker exec -it hotel-postgres psql -U admin -d hotel_db

# Once inside psql, useful commands:
# \dt                    - List all tables
# \d users               - Describe users table
# \d bookings            - Describe bookings table
# \d payments            - Describe payments table
# SELECT * FROM users;   - View all users
# SELECT * FROM bookings; - View all bookings
# SELECT * FROM payments; - View all payments
# \q                     - Quit psql

# Run SQL query from command line
docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT * FROM users;"
docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT * FROM bookings;"
docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT * FROM payments;"

# Count records
docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT COUNT(*) FROM users;"
docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT COUNT(*) FROM bookings;"
docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT COUNT(*) FROM payments;"

# Backup database
docker exec hotel-postgres pg_dump -U admin hotel_db > backup.sql

# Restore database
docker exec -i hotel-postgres psql -U admin -d hotel_db < backup.sql


## ========================================
## MONGODB COMMANDS
## ========================================

# Access MongoDB CLI (mongosh)
docker exec -it hotel-mongo mongosh -u admin -p admin

# Once inside mongosh, useful commands:
# show dbs                           - List all databases
# use hotel_db                       - Switch to hotel_db
# show collections                   - List all collections
# db.hotels.find()                   - View all hotels
# db.hotels.find().pretty()          - View all hotels (formatted)
# db.hotels.countDocuments()         - Count hotels
# db.hotels.find({city: "New York"}) - Find hotels in New York
# exit                               - Quit mongosh

# Run MongoDB query from command line
docker exec -it hotel-mongo mongosh -u admin -p admin --eval "use hotel_db; db.hotels.find().pretty()"
docker exec -it hotel-mongo mongosh -u admin -p admin --eval "use hotel_db; db.hotels.countDocuments()"

# Find hotels by city
docker exec -it hotel-mongo mongosh -u admin -p admin --eval 'use hotel_db; db.hotels.find({city: "New York"}).pretty()'

# Find hotels with rating >= 4.5
docker exec -it hotel-mongo mongosh -u admin -p admin --eval 'use hotel_db; db.hotels.find({rating: {$gte: 4.5}}).pretty()'

# Backup MongoDB
docker exec hotel-mongo mongodump -u admin -p admin --authenticationDatabase admin --out /tmp/backup
docker cp hotel-mongo:/tmp/backup ./mongo-backup

# Restore MongoDB
docker cp ./mongo-backup hotel-mongo:/tmp/backup
docker exec hotel-mongo mongorestore -u admin -p admin --authenticationDatabase admin /tmp/backup


## ========================================
## REDIS COMMANDS
## ========================================

# Access Redis CLI
docker exec -it hotel-redis redis-cli

# Once inside redis-cli, useful commands:
# PING                   - Test connection (should return PONG)
# KEYS *                 - List all keys
# GET key_name           - Get value of a key
# SET key_name value     - Set a key-value pair
# DEL key_name           - Delete a key
# FLUSHALL               - Delete all keys (CAREFUL!)
# INFO                   - Server information
# exit                   - Quit redis-cli

# Run Redis command from command line
docker exec -it hotel-redis redis-cli PING
docker exec -it hotel-redis redis-cli KEYS "*"
docker exec -it hotel-redis redis-cli INFO

# Get specific key
docker exec -it hotel-redis redis-cli GET "some_key"

# Monitor all Redis commands in real-time
docker exec -it hotel-redis redis-cli MONITOR


## ========================================
## KAFKA COMMANDS (Confluent Platform)
## ========================================

# List all topics
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --list

# Create a topic
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --create --topic booking-events --partitions 3 --replication-factor 1
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --create --topic payment-events --partitions 3 --replication-factor 1

# Describe a topic
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --describe --topic booking-events
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --describe --topic payment-events

# Delete a topic
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --delete --topic booking-events

# Produce messages to a topic (interactive)
docker exec -it hotel-kafka kafka-console-producer --bootstrap-server localhost:9092 --topic booking-events
# Type messages and press Enter. Press Ctrl+C to exit.

# Consume messages from a topic (from beginning)
docker exec -it hotel-kafka kafka-console-consumer --bootstrap-server localhost:9092 --topic booking-events --from-beginning

# Consume messages from a topic (only new messages)
docker exec -it hotel-kafka kafka-console-consumer --bootstrap-server localhost:9092 --topic booking-events

# Consume with consumer group
docker exec -it hotel-kafka kafka-console-consumer --bootstrap-server localhost:9092 --topic booking-events --group my-consumer-group

# List consumer groups
docker exec -it hotel-kafka kafka-consumer-groups --bootstrap-server localhost:9092 --list

# Describe consumer group
docker exec -it hotel-kafka kafka-consumer-groups --bootstrap-server localhost:9092 --describe --group my-consumer-group


## ========================================
## ZOOKEEPER COMMANDS
## ========================================

# Check Zookeeper status
docker exec -it hotel-zookeeper zookeeper-shell localhost:2181

# Once inside zookeeper-shell:
# ls /                   - List root znodes
# ls /brokers/ids        - List Kafka broker IDs
# get /brokers/ids/1     - Get broker info
# quit                   - Exit


## ========================================
## RABBITMQ COMMANDS
## ========================================

# Access RabbitMQ Management UI
# Open browser: http://localhost:15672
# Username: admin
# Password: admin

# List queues
docker exec -it hotel-rabbitmq rabbitmqctl list_queues

# List exchanges
docker exec -it hotel-rabbitmq rabbitmqctl list_exchanges

# List bindings
docker exec -it hotel-rabbitmq rabbitmqctl list_bindings

# List connections
docker exec -it hotel-rabbitmq rabbitmqctl list_connections

# List channels
docker exec -it hotel-rabbitmq rabbitmqctl list_channels

# Purge a queue (delete all messages)
docker exec -it hotel-rabbitmq rabbitmqctl purge_queue booking_notifications
docker exec -it hotel-rabbitmq rabbitmqctl purge_queue payment_notifications

# Delete a queue
docker exec -it hotel-rabbitmq rabbitmqctl delete_queue booking_notifications

# Check cluster status
docker exec -it hotel-rabbitmq rabbitmqctl cluster_status

# Check node status
docker exec -it hotel-rabbitmq rabbitmqctl status

## ========================================
## INITIALIZE KAFKA TOPICS
## ========================================

# Create booking-events topic
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --create --topic booking-events --partitions 3 --replication-factor 1 --if-not-exists

# Create payment-events topic
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --create --topic payment-events --partitions 3 --replication-factor 1 --if-not-exists

# Verify topics were created
docker exec -it hotel-kafka kafka-topics --bootstrap-server localhost:9092 --list


## ========================================
## INITIALIZE RABBITMQ QUEUES
## ========================================

# Note: RabbitMQ queues are typically created by the application when it connects.
# However, you can create them manually using rabbitmqadmin (if installed):

# Install rabbitmqadmin (one-time setup)
docker exec -it hotel-rabbitmq wget http://localhost:15672/cli/rabbitmqadmin -O /usr/local/bin/rabbitmqadmin
docker exec -it hotel-rabbitmq chmod +x /usr/local/bin/rabbitmqadmin

# Create queues
docker exec -it hotel-rabbitmq rabbitmqadmin declare queue name=booking_notifications durable=true
docker exec -it hotel-rabbitmq rabbitmqadmin declare queue name=payment_notifications durable=true
