┌─────────────────────────────────────────────────────────────────────────────┐
│ 🗄️  POSTGRESQL COMMANDS                                                     │
└─────────────────────────────────────────────────────────────────────────────┘

CONNECTION DETAILS:
  Host: localhost
  Port: 5432
  Database: hotel_db
  Username: admin
  Password: admin

CONNECT TO PSQL:
  docker exec -it hotel-postgres psql -U admin -d hotel_db

COMMON PSQL COMMANDS:
  \l                          -- List all databases
  \c hotel_db                 -- Connect to database
  \dt                         -- List all tables
  \d table_name               -- Describe table
  \du                         -- List users
  \q                          -- Quit

EXECUTE QUERY FROM TERMINAL:
  docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT * FROM users;"

CREATE USERS TABLE:
  docker exec -it hotel-postgres psql -U admin -d hotel_db -c "
  CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone_number VARCHAR(20),
    role VARCHAR(50) DEFAULT 'CUSTOMER',
    enabled BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
  );"

VIEW ALL USERS:
  docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT * FROM users;"

COUNT RECORDS:
  docker exec -it hotel-postgres psql -U admin -d hotel_db -c "SELECT COUNT(*) FROM users;"

BACKUP DATABASE:
  docker exec -it hotel-postgres pg_dump -U admin hotel_db > backup.sql

RESTORE DATABASE:
  docker exec -i hotel-postgres psql -U admin hotel_db < backup.sql

┌─────────────────────────────────────────────────────────────────────────────┐
│ 🍃 MONGODB COMMANDS                                                         │
└─────────────────────────────────────────────────────────────────────────────┘

CONNECTION DETAILS:
  Host: localhost
  Port: 27017
  Username: admin
  Password: admin

CONNECT TO MONGOSH:
  docker exec -it hotel-mongo mongosh -u admin -p admin

COMMON MONGOSH COMMANDS:
  show dbs                    -- List all databases
  use hotel_db                -- Switch to database
  show collections            -- List collections
  db.hotels.find()            -- Find all documents
  db.hotels.find().pretty()   -- Pretty print
  db.hotels.countDocuments()  -- Count documents
  exit                        -- Exit mongosh

INSERT SAMPLE HOTEL:
  docker exec -it hotel-mongo mongosh -u admin -p admin --eval '
  use hotel_db
  db.hotels.insertOne({
    name: "Grand Plaza Hotel",
    location: "New York",
    rating: 4.5,
    amenities: ["WiFi", "Pool", "Gym"]
  })'

FIND ALL HOTELS:
  docker exec -it hotel-mongo mongosh -u admin -p admin --eval 'use hotel_db; db.hotels.find()'

FIND BY LOCATION:
  docker exec -it hotel-mongo mongosh -u admin -p admin --eval 'use hotel_db; db.hotels.find({location: "New York"})'

COUNT HOTELS:
  docker exec -it hotel-mongo mongosh -u admin -p admin --eval 'use hotel_db; db.hotels.countDocuments()'

DELETE ALL HOTELS:
  docker exec -it hotel-mongo mongosh -u admin -p admin --eval 'use hotel_db; db.hotels.deleteMany({})'

┌─────────────────────────────────────────────────────────────────────────────┐
│ 🔴 REDIS COMMANDS                                                           │
└─────────────────────────────────────────────────────────────────────────────┘

CONNECTION DETAILS:
  Host: localhost
  Port: 6379

CONNECT TO REDIS-CLI:
  docker exec -it hotel-redis redis-cli

COMMON REDIS COMMANDS:
  PING                        -- Test connection
  SET key "value"             -- Set a key
  GET key                     -- Get a key
  KEYS *                      -- List all keys
  DEL key                     -- Delete a key
  EXISTS key                  -- Check if key exists
  TTL key                     -- Get time to live
  EXPIRE key 3600             -- Set expiration (seconds)
  FLUSHALL                    -- Delete all keys (careful!)
  INFO                        -- Server info
  exit                        -- Exit

SET VALUE FROM TERMINAL:
  docker exec -it hotel-redis redis-cli SET mykey "Hello World"

GET VALUE FROM TERMINAL:
  docker exec -it hotel-redis redis-cli GET mykey

SET WITH EXPIRATION:
  docker exec -it hotel-redis redis-cli SETEX session:123 3600 "user_data"

LIST ALL KEYS:
  docker exec -it hotel-redis redis-cli KEYS "*"

MONITOR ALL COMMANDS:
  docker exec -it hotel-redis redis-cli MONITOR

GET MEMORY INFO:
  docker exec -it hotel-redis redis-cli INFO memory

┌─────────────────────────────────────────────────────────────────────────────┐
│ 📨 KAFKA COMMANDS                                                           │
└─────────────────────────────────────────────────────────────────────────────┘

CONNECTION DETAILS:
  Bootstrap Servers (External): localhost:9093
  Bootstrap Servers (Internal): kafka:9092
  Zookeeper: localhost:2181

LIST TOPICS:
  docker exec -it hotel-kafka kafka-topics.sh --bootstrap-server localhost:9092 --list

CREATE TOPIC:
  docker exec -it hotel-kafka kafka-topics.sh \
    --bootstrap-server localhost:9092 \
    --create \
    --topic booking-events \
    --partitions 3 \
    --replication-factor 1

DESCRIBE TOPIC:
  docker exec -it hotel-kafka kafka-topics.sh \
    --bootstrap-server localhost:9092 \
    --describe \
    --topic booking-events

DELETE TOPIC:
  docker exec -it hotel-kafka kafka-topics.sh \
    --bootstrap-server localhost:9092 \
    --delete \
    --topic booking-events

PRODUCE MESSAGES (INTERACTIVE):
  docker exec -it hotel-kafka kafka-console-producer.sh \
    --bootstrap-server localhost:9092 \
    --topic booking-events
  # Type messages and press Enter, Ctrl+C to exit

CONSUME MESSAGES (FROM BEGINNING):
  docker exec -it hotel-kafka kafka-console-consumer.sh \
    --bootstrap-server localhost:9092 \
    --topic booking-events \
    --from-beginning

CONSUME MESSAGES (LATEST ONLY):
  docker exec -it hotel-kafka kafka-console-consumer.sh \
    --bootstrap-server localhost:9092 \
    --topic booking-events

CREATE REQUIRED TOPICS:
  docker exec -it hotel-kafka kafka-topics.sh --bootstrap-server localhost:9092 --create --topic booking-events --partitions 3 --replication-factor 1
  docker exec -it hotel-kafka kafka-topics.sh --bootstrap-server localhost:9092 --create --topic payment-events --partitions 3 --replication-factor 1

  docker exec -it hotel-rabbitmq rabbitmqadmin purge queue \
    name=booking_notifications

  docker exec -it hotel-rabbitmq rabbitmqadmin get queue=booking_notifications

  docker exec -it hotel-rabbitmq rabbitmqadmin publish \
    exchange=amq.default \
    routing_key=booking_notifications \
    payload="Test message"

SERVER STATUS:

CREATE REQUIRED QUEUES:
  docker exec -it hotel-rabbitmq rabbitmqadmin declare queue name=booking_notifications durable=true
